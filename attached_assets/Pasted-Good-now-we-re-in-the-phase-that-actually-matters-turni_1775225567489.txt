def calculate_slippage(order_size_aud, spread_pct, vol_1m, depth_factor):
    base = spread_pct * 0.5

    size_impact = (order_size_aud / 30000) * 0.001
    vol_impact = min(vol_1m * 0.002, 0.003)

    random_noise = abs(__import__("random").gauss(0, 0.0005))

    slippage = base + size_impact + vol_impact + random_noise

    return min(slippage, 0.006)

Add simple tracker

Create common/performance.py:

stats = {
    "trades": 0,
    "wins": 0,
    "losses": 0,
    "pnl": []
}

def record_trade(pnl):
    stats["trades"] += 1
    stats["pnl"].append(pnl)

    if pnl > 0:
        stats["wins"] += 1
    else:
        stats["losses"] += 1

def get_stats():
    if not stats["pnl"]:
        return {}

    avg_win = sum(p for p in stats["pnl"] if p > 0) / max(stats["wins"], 1)
    avg_loss = sum(p for p in stats["pnl"] if p <= 0) / max(stats["losses"], 1)

    return {
        "win_rate": stats["wins"] / stats["trades"],
        "avg_win": avg_win,
        "avg_loss": avg_loss,
        "expectancy": avg_win * 0.5 + avg_loss * 0.5
    }
