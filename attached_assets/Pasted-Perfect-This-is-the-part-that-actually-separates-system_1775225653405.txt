📈 STEP 5 — Simple Edge Analyzer Script

Create:

research/analyse_trades.py

import pandas as pd

df = pd.read_csv("trades.csv")

print("Total Trades:", len(df))
print("Win Rate:", (df["pnl"] > 0).mean())

# By strategy
print("\nBy Strategy:")
print(df.groupby("strategy")["pnl"].mean())

# By confidence
df["conf_band"] = pd.cut(df["confidence"], [0.6, 0.65, 0.7, 0.8])
print("\nBy Confidence:")
print(df.groupby("conf_band")["pnl"].mean())

# By regime
print("\nBy Regime:")
print(df.groupby("regime")["pnl"].mean())

# By hour
print("\nBy Hour:")
print(df.groupby("hour")["pnl"].mean().sort_values())
