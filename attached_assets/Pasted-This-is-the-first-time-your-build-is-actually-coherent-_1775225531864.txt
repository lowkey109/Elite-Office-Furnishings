import httpx

async def process_bar(bar: dict):
    async with httpx.AsyncClient() as client:
        # 1. Data quality
        dq = await client.post("http://localhost:8002/gate", json=bar)
        dq_json = dq.json()
        if not dq_json["passed"]:
            print(f"[BLOCKED] {dq_json['reason']}")
            return

        # 2. Features
        feat = await client.post("http://localhost:8003/process", json=bar)
        feat_json = feat.json()

        # 3. Strategy
        strat = await client.post(
            "http://localhost:8004/generate",
            json={"bar": bar, "features": feat_json["features"]}
        )
        candidates = strat.json()["candidates"]
        if not candidates:
            return

        # 4. AI scoring
        ai = await client.post("http://localhost:8005/score", json=candidates)
        scored = ai.json()["scored_candidates"]

        # 5. Risk + compliance + execution
        for payload in scored:
            rc = await client.post("http://localhost:8006/validate", json=payload)
            approved = rc.json()["approved"]

            exec_res = await client.post(
                "http://localhost:8007/execute",
                json={"payload": payload, "approved": approved}
            )

            # 6. Audit
            await client.post(
                "http://localhost:8008/log",
                json={"payload": payload, "outcome": exec_res.json()}
            )

👉 This is your entire system loop

import asyncio
import httpx
import random
from datetime import datetime

async def send_fake_bars():
    price = 65000

    async with httpx.AsyncClient() as client:
        while True:
            price += random.uniform(-100, 100)

            bar = {
                "symbol": "BTCUSDT",
                "time": datetime.utcnow().isoformat(),
                "open": price,
                "high": price + random.uniform(0, 50),
                "low": price - random.uniform(0, 50),
                "close": price,
                "volume": random.uniform(100, 1000),
                "latency_ms": random.randint(50, 500),
                "spread_pct": random.uniform(0.0003, 0.001),
                "vol_spike": random.uniform(0.5, 2.5),
                "avg_volume_20": 500
            }

            await client.post("http://localhost:8001/bar", json=bar)
            await asyncio.sleep(2)

if __name__ == "__main__":
    asyncio.run(send_fake_bars())
