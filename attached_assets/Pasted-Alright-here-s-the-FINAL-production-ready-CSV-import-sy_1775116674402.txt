🧱 1. BACKEND — SERVICE (CORE LOGIC)

server/services/leadCsvImportService.ts

import type { InsertLead } from "@shared/schema";

type PreviewRow = {
  row: number;
  data: Partial<InsertLead>;
  status: "valid" | "invalid" | "duplicate";
  reason?: string;
};

type PreviewResult = {
  valid: PreviewRow[];
  invalid: PreviewRow[];
  duplicates: PreviewRow[];
};

function parseCSV(csv: string) {
  const rows = csv.split("\n").map(r => r.trim()).filter(Boolean);
  const headers = rows.shift()?.split(",").map(h => h.trim()) || [];

  return rows.map((row, i) => {
    const values = row.split(",");
    const obj: any = {};
    headers.forEach((h, idx) => obj[h] = values[idx]?.trim());
    return { row: i + 1, obj };
  });
}

function mapToLead(obj: any): Partial<InsertLead> {
  return {
    type: "csv_import",
    name: obj.contactName || "Unknown",
    email: obj.email,
    company: obj.companyName || "",
    phone: obj.phone || "",
    officeLocation: obj.city || "",
    message: obj.notes || "",
    sourcePage: "csv_import",
    leadStatus: "new",

    staffCount: obj.staffCount ? parseInt(obj.staffCount) : undefined,
    budgetMin: obj.budgetMin ? parseInt(obj.budgetMin) : undefined,
    budgetMax: obj.budgetMax ? parseInt(obj.budgetMax) : undefined,
    estimatedValueMin: obj.estimatedValue ? parseInt(obj.estimatedValue) : undefined,
    estimatedValueMax: obj.estimatedValue ? parseInt(obj.estimatedValue) : undefined,
  };
}

export async function previewCSV(csv: string, storage: any): Promise<PreviewResult> {
  const parsed = parseCSV(csv);

  const result: PreviewResult = {
    valid: [],
    invalid: [],
    duplicates: [],
  };

  for (const { row, obj } of parsed) {
    const lead = mapToLead(obj);

    if (!lead.email || !lead.name) {
      result.invalid.push({ row, data: lead, status: "invalid", reason: "Missing name/email" });
      continue;
    }

    if (!lead.email.includes("@")) {
      result.invalid.push({ row, data: lead, status: "invalid", reason: "Invalid email" });
      continue;
    }

    const exists = await storage.findLeadByEmail?.(lead.email);

    if (exists) {
      result.duplicates.push({ row, data: lead, status: "duplicate" });
      continue;
    }

    result.valid.push({ row, data: lead, status: "valid" });
  }

  return result;
}

export async function importCSV(csv: string, storage: any) {
  const preview = await previewCSV(csv, storage);

  let imported = 0;

  for (const row of preview.valid) {
    await storage.createLead(row.data);
    imported++;
  }

  return {
    imported,
    duplicates: preview.duplicates.length,
    invalid: preview.invalid.length,
  };
}

🧱 2. BACKEND — ROUTES

server/routes/leadCsvImportRoute.ts

import express from "express";
import { previewCSV, importCSV } from "../services/leadCsvImportService";

export const csvImportRouter = express.Router();

// Preview endpoint
csvImportRouter.post("/preview-csv", async (req, res) => {
  try {
    const { csv } = req.body;
    const result = await previewCSV(csv, req.app.locals.storage);
    res.json(result);
  } catch (e) {
    res.status(500).json({ error: "Preview failed" });
  }
});

// Import endpoint
csvImportRouter.post("/import-csv", async (req, res) => {
  try {
    const { csv } = req.body;
    const result = await importCSV(csv, req.app.locals.storage);
    res.json(result);
  } catch (e) {
    res.status(500).json({ error: "Import failed" });
  }
});

🎯 4. FRONTEND — REAL ADMIN UI

client/src/components/admin/CSVImportModal.tsx

import React, { useState } from "react";

export default function CSVImportModal({ onClose, onSuccess }: any) {
  const [csvText, setCsvText] = useState("");
  const [preview, setPreview] = useState<any>(null);
  const [loading, setLoading] = useState(false);

  const handlePreview = async () => {
    setLoading(true);

    const res = await fetch("/api/leads/preview-csv", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ csv: csvText }),
    });

    const data = await res.json();
    setPreview(data);
    setLoading(false);
  };

  const handleImport = async () => {
    setLoading(true);

    const res = await fetch("/api/leads/import-csv", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ csv: csvText }),
    });

    const data = await res.json();

    alert(`Imported: ${data.imported}`);
    onSuccess?.();
    onClose();
  };

  return (
    <div style={{position:"fixed",top:0,left:0,right:0,bottom:0,background:"rgba(0,0,0,0.8)"}}>
      <div style={{maxWidth:700,margin:"60px auto",background:"#111",padding:20,color:"#fff"}}>

        <h2>CSV Import</h2>

        <textarea
          rows={10}
          value={csvText}
          onChange={(e) => setCsvText(e.target.value)}
          style={{width:"100%",background:"#222",color:"#fff"}}
        />

        <div style={{marginTop:10}}>
          <button onClick={handlePreview}>Preview</button>
          {preview && <button onClick={handleImport} style={{marginLeft:10}}>Import</button>}
          <button onClick={onClose} style={{marginLeft:10}}>Cancel</button>
        </div>

        {preview && (
          <div style={{marginTop:20}}>
            <p>✅ Valid: {preview.valid.length}</p>
            <p>⚠️ Invalid: {preview.invalid.length}</p>
            <p>🔁 Duplicates: {preview.duplicates.length}</p>
          </div>
        )}

      </div>
    </div>
  );
}
