import pandas as pd
import itertools

df = pd.read_csv("trades.csv")

print("========== OVERVIEW ==========")
print("Total Trades:", len(df))
print("Win Rate:", (df["pnl"] > 0).mean())
print("Avg PnL:", df["pnl"].mean())

# Basic segmentation
print("\n========== BY STRATEGY ==========")
print(df.groupby("strategy")["pnl"].agg(["count","mean"]))

print("\n========== BY REGIME ==========")
print(df.groupby("regime")["pnl"].agg(["count","mean"]))

print("\n========== BY HOUR ==========")
print(df.groupby("hour")["pnl"].mean().sort_values())

# Confidence bands
df["conf_band"] = pd.cut(df["confidence"], [0.6, 0.65, 0.7, 0.75, 0.8])
print("\n========== BY CONFIDENCE ==========")
print(df.groupby("conf_band")["pnl"].agg(["count","mean"]))

# Volume bands
df["vol_band"] = pd.cut(df["volume_ratio"], [0, 1.2, 1.5, 2.0])
print("\n========== BY VOLUME ==========")
print(df.groupby("vol_band")["pnl"].agg(["count","mean"]))

# -------------------------------------------------------
# 🔥 EDGE FINDER (THIS IS THE GOLD)
# -------------------------------------------------------

print("\n========== TOP EDGE COMBINATIONS ==========")

filters = [
    ("strategy", df["strategy"].unique()),
    ("regime", df["regime"].unique()),
    ("hour", sorted(df["hour"].unique())),
]

best_edges = []

for strat in df["strategy"].unique():
    for regime in df["regime"].unique():
        for hour in df["hour"].unique():
            subset = df[
                (df["strategy"] == strat) &
                (df["regime"] == regime) &
                (df["hour"] == hour)
            ]

            if len(subset) < 5:
                continue

            avg_pnl = subset["pnl"].mean()
            win_rate = (subset["pnl"] > 0).mean()

            if avg_pnl > 0:
                best_edges.append({
                    "strategy": strat,
                    "regime": regime,
                    "hour": hour,
                    "trades": len(subset),
                    "avg_pnl": round(avg_pnl, 2),
                    "win_rate": round(win_rate, 2)
                })

edges_df = pd.DataFrame(best_edges).sort_values(by="avg_pnl", ascending=False)

print(edges_df.head(10))
